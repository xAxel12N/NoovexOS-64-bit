static int ok=1; static void chk(const char* w,int c){ printf("  %-60s %s\n",w,c?"ok":"FAIL"); if(!c)ok=0; }
int main(void){
  add(0,0x9F000,7); add(0x9F000,0x100000,0); add(0x100000,0x09D1F000,7); add(0x09D1F000,0x0A000000,0); add(0x0A000000,0x0A200000,7);

  u64 bs,be; u32 bt;
  chk("bss 0xC00000-0x093CA0A0 is usable", !mm_first_bad(0xC00000,0x093CA0A0,&bs,&be,&bt));

  chk("ACPI NVS reported", mm_first_bad(0x0A000000,0x0A300000,&bs,&be,&bt)&&bs==0x0A200000&&bt==10);
  chk("DMA arena skips the reserved first 128 KB", mm_skip_bad(0x0B000000,4096,0x0C000000)==0x0B020000);
  chk("usable end from 256 MB is 0xC0000000", mm_usable_end(0x10000000)==0xC0000000ULL);
  chk("gap (not in map) detected", (g_mmr[1].t=7, mm_usable_end(0x9F000)==0x09D1F000));
